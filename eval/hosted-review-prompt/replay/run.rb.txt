#!/usr/bin/env ruby
# Replays the review prompts on frozen commits. Each run gets a fresh tree and opaque id.
require 'json'; require 'fileutils'; require 'open3'; require 'securerandom'; require 'optparse'; require 'time'
o = { reps: 1, arms: %w[p q], jobs: 4, bin: 'claude', model: 'claude-sonnet-5-5', cap: 40.0, only: nil, out: 'out' }
OptionParser.new do |x|
  x.on('--reps N', Integer) { o[:reps] = _1 }
  x.on('--arms A') { o[:arms] = _1.split(',') }
  x.on('--jobs N', Integer) { o[:jobs] = _1 }
  x.on('--bin B') { o[:bin] = _1 }
  x.on('--model M') { o[:model] = _1 }
  x.on('--cap USD', Float) { o[:cap] = _1 }
  x.on('--only LIST') { o[:only] = _1.split(',') }
  x.on('--out D') { o[:out] = _1 }
end.parse!
root = File.expand_path(__dir__)
out = "#{root}/#{o[:out]}"; FileUtils.mkdir_p(out)
map_path = "#{out}/map.jsonl"
done = File.exist?(map_path) ? File.readlines(map_path).map { JSON.parse(_1) } : []
shas = Dir["#{root}/state/*"].map { File.basename(_1) }
shas = shas.select { |s| o[:only].any? { s.start_with?(_1) } } if o[:only]
todo = shas.flat_map do |sha|
  o[:arms].flat_map do |arm|
    next [] unless File.exist?("#{root}/state/#{sha}/prompt-#{arm}.txt")
    have = done.count { _1['sha'] == sha && _1['arm'] == arm && _1['ok'] }
    (have...o[:reps]).map { |i| { sha:, arm:, rep: i } }
  end
end.shuffle(random: Random.new(640))
puts "planned=#{todo.size}"
mcp = "#{root}/tools/mcp.json"
File.write(mcp, { mcpServers: { github_inline_comment: { command: RbConfig.ruby, args: ["#{root}/tools/inline.rb"] } } }.to_json)
allowed = 'mcp__github_inline_comment__create_inline_comment,Bash(gh pr comment:*),Bash(gh pr diff:*),Bash(gh pr view:*)'
lock = Mutex.new
spent = done.sum { _1['cost'].to_f }
q = Queue.new; todo.each { q << _1 }
workers = Array.new(o[:jobs]) do
  Thread.new do
    while (t = (q.pop(true) rescue nil))
      stop = lock.synchronize { spent >= o[:cap] }
      if stop then puts 'CAP reached, stopping'; break end
      id = SecureRandom.hex(4)
      run_dir = "#{out}/#{id}"; FileUtils.mkdir_p(run_dir)
      state = "#{root}/state/#{t[:sha]}"
      wk = "#{root}/wk/#{id}/shaka"; FileUtils.mkdir_p(wk)
      system('tar', '-xf', "#{state}/tree.tar", '-C', wk, exception: true)
      system("cd '#{wk}' && git init -q && git add -A && git -c user.name=github-actions -c user.email=a@example.com commit -qm 'Merge into base' --no-gpg-sign", exception: true)
      env = { 'HOME' => Dir.home, 'USER' => ENV['USER'], 'LANG' => 'en_US.UTF-8', 'TMPDIR' => ENV['TMPDIR'], 'SHELL' => '/bin/bash',
              'PATH' => "#{root}/tools/bin:#{ENV['PATH']}", 'PR_STATE' => state, 'PR_OUT' => run_dir,
              'GH_TOKEN' => 'invalid', 'GH_ENTERPRISE_TOKEN' => 'invalid', 'GH_HOST' => 'invalid.localhost', 'GH_CONFIG_DIR' => "#{root}/tools/ghcfg",
              'CLAUDE_CODE_DISABLE_AUTO_MEMORY' => '1' }
      cmd = [o[:bin], '-p', '--model', o[:model], '--output-format', 'json', '--setting-sources', 'project', '--strict-mcp-config', '--mcp-config', mcp,
             '--no-session-persistence', '--max-budget-usd', '2', '--allowedTools', allowed]
      started = Time.now
      so, se, st = Open3.capture3(env, *cmd, stdin_data: File.read("#{state}/prompt-#{t[:arm]}.txt"), chdir: wk, unsetenv_others: true)
      File.write("#{run_dir}/stdout.json", so); File.write("#{run_dir}/stderr.txt", se)
      res = begin JSON.parse(so) rescue {} end
      comments = File.exist?("#{run_dir}/comments.jsonl") ? File.readlines("#{run_dir}/comments.jsonl").size : 0
      rec = { id:, sha: t[:sha], arm: t[:arm], rep: t[:rep], exit: st.exitstatus, cost: res['total_cost_usd'], turns: res['num_turns'], ms: res['duration_ms'], denials: (res['permission_denials'] || []).size,
              is_error: res['is_error'], comments:, inline: File.exist?("#{run_dir}/inline.jsonl") ? File.readlines("#{run_dir}/inline.jsonl").size : 0,
              ok: st.success? && res['is_error'] == false && comments.positive?, wall: (Time.now - started).round }
      lock.synchronize do
        spent += res['total_cost_usd'].to_f
        File.open(map_path, 'a') { _1.puts(rec.to_json) }
        puts "#{t[:sha][0, 8]} #{t[:arm]}#{t[:rep]} id=#{id} ok=#{rec[:ok]} cost=#{rec[:cost]} turns=#{rec[:turns]} inline=#{rec[:inline]} spent=#{spent.round(3)}"
      end
      system('rm', '-rf', "#{root}/wk/#{id}")
    end
  end
end
workers.each(&:join)
puts "spent_total=#{spent.round(3)}"
